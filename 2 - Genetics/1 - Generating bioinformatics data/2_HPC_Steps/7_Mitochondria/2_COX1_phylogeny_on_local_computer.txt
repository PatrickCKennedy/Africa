# To focus on high-quality data, we will exclude from the phylogeny all wasps that have an average coverage across the COX1 region of less than 10X, using the 'coverage_summary.txt' file.

my_THRESHOLD_COVERAGE=10
cd /Volumes/Social_evolution_in_animals/COX1_genes_extracted_with_Samtools
awk -v t="$my_THRESHOLD_COVERAGE" '$NF >= t {print $1}' coverage_summary.txt | xargs -I{} cp {}* /Volumes/Social_evolution_in_animals/COX1_genes_extracted_with_Samtools/1_The_subset_above_10X_coverage_of_COX1/

cd /Volumes/Social_evolution_in_animals/COX1_genes_extracted_with_Samtools
my_THRESHOLD_COVERAGE=10
awk -v t="$my_THRESHOLD_COVERAGE" '$NF >= t {print $1}' coverage_summary.txt | while read prefix; do
    cp *"${prefix}"* /Volumes/Social_evolution_in_animals/COX1_genes_extracted_with_Samtools/1_The_subset_above_10X_coverage_of_COX1/
done
