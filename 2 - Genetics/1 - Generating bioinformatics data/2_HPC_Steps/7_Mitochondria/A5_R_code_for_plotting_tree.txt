 # Define the two species clades:
                            cladeA <- c("LouwsFarm_PETIOLATA_WASPS_L9_h",
                                        "LouwsFarm_PETIOLATA_WASPS_L12_c")
                            cladeB <- c("Somalomo_BRIS_EF44_EGG6",

                            # Get MRCA nodes
                            nodeA <- getMRCA(tree, cladeA)
                            nodeB <- getMRCA(tree, cladeB)
                            
                            # Midpoint reroot between these nodes
                            midpoint_root <- function(tree, nodeA, nodeB) {
                              path <- nodepath(tree, nodeA, nodeB)
                              edges <- sapply(seq_along(path[-length(path)]), function(i)
                                which(apply(tree$edge, 1, function(e) all(e %in% path[i:(i+1)]))))
                              lengths <- ifelse(is.null(tree$edge.length), rep(1, length(edges)), tree$edge.length[edges])
                              half <- sum(lengths) / 2
                              cum <- 0
                              for (i in seq_along(edges)) {
                                cum <- cum + lengths[i]
                                if (cum >= half) {
                                  child <- tree$edge[edges[i], 2]
                                  pos <- cum - half
                                  return(reroot(tree, node = child, position = pos))
                                }
                              }
                            }
                            
                            tree_rooted <- midpoint_root(tree, nodeA, nodeB)
                            tree_rooted_and_ladderised <- ladderize(tree_rooted)
                        

# Simple plot:
plot(tree_rooted_and_ladderised, show.node.label = TRUE)

# ggtree plot:
library(ggtree)
library(ggplot2)

# Extract tip labels:
tips <- tree_rooted_and_ladderised$tip.label

# Make a table containing the population that each wasp comes from (given by the prefix in its name):
metadata <- data.frame(
  label = tips,
  population = sapply(strsplit(tips, "_"), `[`, 1)
)
View(metadata)

# Add the climate zone information:
library(dplyr)
metadata <- metadata %>%
  mutate(climate = case_when(
    population %in% c("AfricanLaboratory", "PeterRobertsFarm", "Vusumuzi", "Cowshed", "LoversRock", "TrudiesFarm", "Pumphouse", "ZebraHouse") ~ "South_Africa",
    population %in% c("Soa", "Yaounde1", "Somalomo", "Mbouda", "Douala", "Messa", "Population") ~ "Southern_Cameroon",
    population %in% c("LouwsFarm") ~ "Petiolata_LouwsFarm",
    population == "Maroua" ~ "Northern_Cameroon",
    TRUE ~ NA_character_  # For any populations not listed
  ))

View(metadata)

# Add the bootstrap supports:
# Since there are two values (SHaLRT test and UFboot, take the UFboot values - the second position):

# Parse UFBoot robustly from the tree’s node labels
raw <- tree_rooted_and_ladderised$node.label
ufboot <- ifelse(grepl("/", raw), sub("^.*?/", "", raw), raw)
ufboot <- gsub("[^0-9.]", "", ufboot)
ufboot <- suppressWarnings(as.numeric(ufboot))
if (any(!is.na(ufboot)) && max(ufboot, na.rm = TRUE) <= 1) ufboot <- round(ufboot * 100)


# Attach a 'boot' column to internal nodes in p$data (tips get NA)
p$data <- p$data %>%
  mutate(boot = ifelse(!isTip, ufboot, NA_real_))

# Now use 'boot' in aes(), avoiding as.numeric() to prevent coercion across tips
p_boots_included <- p + geom_text2(
  aes(subset = !isTip & !is.na(boot) & boot >= 70,
      label  = round(boot),
      angle  = angle),
  size = 1, vjust = -0.2
)
