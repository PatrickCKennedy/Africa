# I have made a text file with the full paths to all of the BAM files, and call it my_bam_list_fullpaths.txt

# I have also made a text file with the names of the 141 BAM files coming from wasp main samples with
# a raw sequencing depth of at least 3X:
/user/work/pk14885/List_of_wasp_main_samples_with_more_than_3X_raw_sequencing_coverage.txt

# Extract the paths of the 141 target BAM files: 
cd /user/work/pk14885/Wasps/Completed_alignments
grep -F -f /user/work/pk14885/List_of_wasp_main_samples_with_more_than_3X_raw_sequencing_coverage.txt my_bam_list_fullpaths.txt > subset_paths.txt

# To aid computation, we take N random wasps from each population (or all wasps from the population if there are fewer than 10 available) from the 141 wasps. To take all wasps, just set this to something very high (e.g, 999), which makes it irrelevant:

while read -r f; do
  pop=$(basename "$(dirname "$f")")
  printf '%s\t%s\n' "$pop" "$f"
done < subset_paths.txt \
| sort -k1,1 \
| awk -F'\t' '{print > ("pop_"$1".lst")}'  # split into one file per population
for pf in pop_*.lst; do
  shuf -n 999 "$pf" 2>/dev/null
done > subsampled_bams.txt
cut -f2- subsampled_bams.txt > 3X_subsampled_bams_paths_only.txt

# That results in 141 target wasps if you take all wasps from each population at a raw sequencing depth of at least 3X.

# MAKING SYMLINK FOLDER:

# Before running this script as a job on BluePebble, we now make sure all the BAM files (and their
# associated BAI files) are in the same symlink folder, using the following:
# First, make a symlink folder to point towards the BAM files:
                  # If the folder already exists, remove it and make it anew:
                  rm -r /user/work/pk14885/Wasps/Completed_alignments/all_bams_symlink_folder
                  mkdir -p /user/work/pk14885/Wasps/Completed_alignments/all_bams_symlink_folder
                  cd /user/work/pk14885/Wasps/Completed_alignments
  
                  while IFS= read -r bam; do
                      [ -z "$bam" ] && continue
                      [ ! -f "$bam" ] && { echo "WARN: BAM not found: $bam" >&2; continue; }
                  
                      bam_base="$(basename "$bam")"
                      ln -s -f "$bam" "/user/work/pk14885/Wasps/Completed_alignments/all_bams_symlink_folder/$bam_base"
                  
                      # Attempt to locate index files
                      idx1="${bam}.bai"          # e.g., /path/sample.bam.bai
                      idx2="${bam%.*}.bai"       # e.g., /path/sample.bai
                  
                      if [ -f "$idx1" ]; then
                          ln -s -f "$idx1" "/user/work/pk14885/Wasps/Completed_alignments/all_bams_symlink_folder/${bam_base}.bai"
                      elif [ -f "$idx2" ]; then
                          ln -s -f "$idx2" "/user/work/pk14885/Wasps/Completed_alignments/all_bams_symlink_folder/${bam_base%.bam}.bai"
                      else
                          echo "INFO: No .bai found for $bam" >&2
                      fi
                  done < 3X_subsampled_bams_paths_only.txt

# -----------------------------------------------------------------------------
# MAKE A LIST OF ALL THE PAIRS OF GENOMES THAT distANGSD WILL CONSIDER:
# As above, run this section once (manually) before submitting the array job.

my_BAM_directory="/user/work/pk14885/Wasps/Completed_alignments/all_bams_symlink_folder"
my_output_directory="${my_BAM_directory}/distangsd_results"
pair_list="${my_output_directory}/pairs.tsv"

mkdir -p "$my_output_directory"

# Sort out the BAM files into order:
cd /user/work/pk14885/Wasps/Completed_alignments
mapfile -t BAMS < <(find -L "$my_BAM_directory" -maxdepth 1 -type f -name "*.bam" | sort)

# Identify all the pairs that need to be run (this takes a bit of time to complete):
cd /user/work/pk14885/Wasps/Completed_alignments
: > "$pair_list"
total=${#BAMS[@]}
for ((i=0; i<total; i++)); do
  # Print progress every 100 iterations (or every iteration if you prefer)
  printf "Progress: %d of %d BAMs processed (%.2f%%)\n" "$i" "$total" "$(bc <<< "scale=2; $i/$total*100")"

  for ((j=i+1; j<total; j++)); do
    A="${BAMS[$i]}" ; B="${BAMS[$j]}"
    a_base="$(basename "${A%.bam}")"
    b_base="$(basename "${B%.bam}")"
    echo -e "${A}\t${B}\t${my_output_directory}/${a_base}_vs_${b_base}" >> "$pair_list"
  done
done
